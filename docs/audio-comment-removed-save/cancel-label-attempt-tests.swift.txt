    @MainActor private func audioReveal(_ element: XCUIElement, _ app: XCUIApplication, towardTop: Bool = false) {
        let navigation = app.navigationBars["Audio draft"], form = app.collectionViews["audioDraftForm"]
        XCTAssertTrue(form.waitForExistence(timeout: 8), app.debugDescription)
        func viewport() -> CGRect {
            let frame = form.frame.intersection(app.windows.firstMatch.frame)
            let top = max(frame.minY, navigation.frame.maxY)
            let probe = app.staticTexts["audioFixtureLookupProbe"]
            let bottom = min(frame.maxY, probe.exists ? probe.frame.minY - 16 : frame.maxY) - 4
            return CGRect(x: frame.minX + 4, y: top + 4, width: frame.width - 8, height: max(0, bottom - top - 4))
        }
        func visible() -> Bool {
            element.exists && element.isHittable && !element.frame.isEmpty && viewport().contains(element.frame)
        }
        for attempt in 0..<24 {
            if visible() { break }
            let area = viewport()
            XCTAssertGreaterThan(area.height, 0, app.debugDescription)
            let down = element.exists && !element.frame.isEmpty ? element.frame.midY < area.midY : (attempt < 12 ? towardTop : !towardTop)
            let x = area.minX + 24 - app.frame.minX
            let from = area.minY + area.height * (down ? 0.25 : 0.8) - app.frame.minY
            let to = area.minY + area.height * (down ? 0.8 : 0.25) - app.frame.minY
            let origin = app.coordinate(withNormalizedOffset: .zero)
            origin.withOffset(CGVector(dx: x, dy: from)).press(forDuration: 0.05, thenDragTo: origin.withOffset(CGVector(dx: x, dy: to)), withVelocity: .slow, thenHoldForDuration: 0.1)
        }
        if !visible() { audioScreenshot(app, "Audio control is outside the actual form viewport") }
        XCTAssertTrue(visible(), "Complete audio control must fit above native and fixture overlays: " + app.debugDescription)
    }
